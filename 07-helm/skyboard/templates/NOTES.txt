  kubectl -n {{ .Release.Namespace }} rollout status deployment/{{ include "skyboard.fullname" . }}-api

{{- if .Values.ingress.enabled }}

{{- else }}

  kubectl -n {{ .Release.Namespace }} port-forward svc/{{ include "skyboard.fullname" . }}-api 8000:8000

{{- end }}
